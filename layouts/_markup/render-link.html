{{- /* Link render hook: renders the link and, for GitHub/GitLab repositories, organisations and groups, appends activity information fetched at build time. Only http(s), mailto and in-page/relative links are emitted as-is; any other scheme (e.g. javascript:) is neutralised. */ -}} {{- $dest := .Destination -}} {{- $u := urls.Parse $dest -}} {{- $scheme := lower $u.Scheme -}} {{- $external := in (slice "http" "https") $scheme -}} {{- $href := "#" -}} {{- if or $external (eq $scheme "mailto") (eq $scheme "") -}}{{- $href = $dest -}}{{- end -}} {{ .Text }} {{- if and $external (not (in site.Params.noActivity (strings.TrimSuffix "/" $dest))) -}} {{- $act := partial "activity.html" $u -}} {{- if and $act $act.missing -}} not found {{- else if $act -}} {{- with $act -}} {{- $days := div (now.Sub .date).Hours 24 | math.Floor | int -}} {{- $cls := "old" -}} {{- if .archived -}}{{- $cls = "archived" -}} {{- else if le $days 183 -}}{{- $cls = "fresh" -}} {{- else if le $days 730 -}}{{- $cls = "aging" -}} {{- end -}} {{- $ago := "today" -}} {{- if ge $days 730 -}}{{- $ago = printf "%d years ago" (div $days 365) -}} {{- else if ge $days 60 -}}{{- $ago = printf "%d months ago" (div $days 30) -}} {{- else if ge $days 14 -}}{{- $ago = printf "%d weeks ago" (div $days 7) -}} {{- else if ge $days 2 -}}{{- $ago = printf "%d days ago" $days -}} {{- else if eq $days 1 -}}{{- $ago = "yesterday" -}} {{- end -}} {{- $title := printf "%s: %s" .label (.date.Format "2 January 2006") -}} {{ $ago }} {{- with .stars -}} {{- $s := cond (ge . 1000) (printf "%.1fk" (div (float .) 1000)) (printf "%d" (int .)) -}} ★ {{ $s }} {{- end -}} {{- with .license -}} {{- if eq . "none" -}}no licence {{- else if eq . "other" -}}other licence {{- else if in . "-ND" -}}{{ . }} · no derivatives {{- else if hasPrefix . "CC-BY-NC" -}}{{ . }} · non-commercial {{- else -}}{{ . }} {{- end -}} {{- end -}} {{- if .sampled -}} {{- $n := .sampled -}} {{- $free := slice -}} {{- range .licenses -}}{{- if in . "-ND" -}}{{- else -}}{{- $free = $free | append . -}}{{- end -}}{{- end -}} {{- $title := printf "Licences of its %d most-starred repositories (forks and archived excluded)" $n -}} {{- if $free -}} {{ delimit $free " · " }}{{ with .unlicensed }} · {{ . }} unlicensed{{ end }} {{- else -}} no licence {{- end -}} {{- end -}} {{- if .archived -}}archived{{- end -}} {{- end -}} {{- end -}} {{- end -}}
