{{- /* Look up repository activity for a parsed URL (url.URL). Returns dict "date" "label" "stars" "archived" "license" , (license only for single repositories/projects) or dict "missing" true when the repository/organisation returns 404, or "" when the host is unsupported or the lookup fails. github.com/owner/repo -> repo metadata + date of the latest commit github.com/owner -> most recently pushed repository of the org/user gitlab.com/group/proj -> project metadata (last_activity_at) gitlab.com/group -> most recently active project of the group/user */ -}} {{- $u := . -}} {{- $out := "" -}} {{- $path := strings.Trim $u.Path "/" -}} {{- $segs := cond (eq $path "") (slice) (split $path "/") -}} {{- $host := lower $u.Host | strings.TrimPrefix "www." -}} {{- if and (eq $host "github.com") (ge (len $segs) 1) (le (len $segs) 2) -}} {{- $h := dict "Accept" "application/vnd.github+json" "X-GitHub-Api-Version" "2022-11-28" -}} {{- with os.Getenv "HUGO_GITHUB_TOKEN" -}} {{- $h = merge $h (dict "Authorization" (printf "Bearer %s" .)) -}} {{- end -}} {{- $api := "https://api.github.com" -}} {{- if eq (len $segs) 2 -}} {{- $repo := partial "fetchjson.html" (dict "url" (printf "%s/repos/%s/%s" $api (index $segs 0) (index $segs 1)) "headers" $h) -}} {{- if and (reflect.IsMap $repo) (isset $repo "_missing") -}} {{- $out = dict "missing" true -}} {{- else if $repo -}} {{- $date := $repo.pushed_at -}} {{- $commits := partial "fetchjson.html" (dict "url" (printf "%s/repos/%s/commits?per_page=1" $api $repo.full_name) "headers" $h) -}} {{- if reflect.IsSlice $commits -}}{{- range first 1 $commits -}}{{- $date = .commit.committer.date -}}{{- end -}}{{- end -}} {{- $lic := "none" -}} {{- with $repo.license -}}{{- $lic = cond (eq .spdx_id "NOASSERTION") "other" .spdx_id -}}{{- end -}} {{- with $date -}} {{- $out = dict "date" (time.AsTime .) "label" "Last commit" "stars" $repo.stargazers_count "archived" $repo.archived "license" $lic -}} {{- end -}} {{- end -}} {{- else -}} {{- $repos := partial "fetchjson.html" (dict "url" (printf "%s/users/%s/repos?sort=pushed&per_page=1" $api (index $segs 0)) "headers" $h) -}} {{- if and (reflect.IsMap $repos) (isset $repos "_missing") -}} {{- $out = dict "missing" true -}} {{- else if reflect.IsSlice $repos -}}{{- range first 1 $repos -}}{{- with .pushed_at -}} {{- $out = dict "date" (time.AsTime .) "label" "Latest activity across its repositories" "stars" nil "archived" false -}} {{- end -}}{{- end -}}{{- end -}} {{- end -}} {{- else if and (eq $host "gitlab.com") (ge (len $segs) 1) -}} {{- $h := dict "Accept" "application/json" -}} {{- $api := "https://gitlab.com/api/v4" -}} {{- if ge (len $segs) 2 -}} {{- $p := partial "fetchjson.html" (dict "url" (printf "%s/projects/%s?license=true" $api (urlquery $path)) "headers" $h) -}} {{- if and (reflect.IsMap $p) (isset $p "_missing") -}} {{- $out = dict "missing" true -}} {{- else if $p -}} {{- $lic := "none" -}} {{- with $p.license -}}{{- $lic = .nickname | default .name | default "other" -}}{{- end -}} {{- with $p.last_activity_at -}} {{- $out = dict "date" (time.AsTime .) "label" "Last activity" "stars" $p.star_count "archived" $p.archived "license" $lic -}} {{- end -}} {{- end -}} {{- else -}} {{- $q := "projects?order_by=last_activity_at&sort=desc&per_page=1" -}} {{- $list := partial "fetchjson.html" (dict "url" (printf "%s/groups/%s/%s&include_subgroups=true" $api (urlquery $path) $q) "headers" $h) -}} {{- if not (reflect.IsSlice $list) -}} {{- $list = partial "fetchjson.html" (dict "url" (printf "%s/users/%s/%s" $api (urlquery $path) $q) "headers" $h) -}} {{- end -}} {{- if and (reflect.IsMap $list) (isset $list "_missing") -}} {{- $out = dict "missing" true -}} {{- else if reflect.IsSlice $list -}}{{- range first 1 $list -}}{{- with .last_activity_at -}} {{- $out = dict "date" (time.AsTime .) "label" "Latest activity across its projects" "stars" nil "archived" false -}} {{- end -}}{{- end -}}{{- end -}} {{- end -}} {{- end -}} {{- return $out -}}
